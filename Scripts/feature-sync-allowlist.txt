# --- Backup -----------------------------------------------------------------

# "Last saved" describes this Mac, not the person's money; a restore must not
# overwrite the date of the backup the person just made.
backup:SettingsStore.lastBackupAt        # machine-local timestamp, deliberately not restored
backup:LocalIntegration.active           # transient: the HTTP server is running right now
backup:LocalIntegration.failure          # transient: last server error text, not a preference

# --- Local API --------------------------------------------------------------

# Backup and restore are file operations on this Mac, not HTTP data mutations.
# See the limitations section in Docs/Backup.md for why the API has no file paths.
# The actions already provide full read and write access to the data.

# Server-assigned metadata and ordering. Clients never set these, and the API
# already exposes ordering through the order of its lists.
api:Account.createdAt                    # server-assigned creation date
api:Account.sortIndex                    # sidebar order is applied server-side
api:Category.createdAt                   # server-assigned creation date
api:Category.sortIndex                   # category order is applied server-side
api:Operation.createdAt                  # server-assigned creation date
api:PlannedOperation.createdAt           # server-assigned creation date

# Coordinates are a UI convenience; the API accepts a human-readable place.
api:Operation.latitude                   # location is UI-only; API accepts `place` text
api:Operation.longitude                  # location is UI-only; API accepts `place` text

# The weekly mask is stored as an integer but serialized as a `weekdays` array,
# which is what a client actually reads and writes.
api:PlannedOperation.weekdayMask         # exposed as `weekdays` in planJSON/create_plan
